cmake_minimum_required(VERSION 3.22.1)
project(karirmate_offline_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
  message(FATAL_ERROR "llama.cpp source missing: copy compatible pinned source to app/src/main/cpp/llama.cpp before enabling native build")
endif()
add_subdirectory(llama.cpp EXCLUDE_FROM_ALL)
add_library(karirmate_llama SHARED llama_jni.cpp)
target_link_libraries(karirmate_llama PRIVATE llama android log)
target_include_directories(karirmate_llama PRIVATE llama.cpp/include)
